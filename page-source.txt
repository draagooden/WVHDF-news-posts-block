    function updatePlaceholders() {
        // Handle default cards grid
        const defaultGrids = document.querySelectorAll('.section-container__cards-grid');

        defaultGrids.forEach(grid => {
            // Remove existing placeholders
            const existingPlaceholders = grid.querySelectorAll('.section-container__card--placeholder');
            existingPlaceholders.forEach(placeholder => placeholder.remove());

            // Get real cards
            const realCards = grid.querySelectorAll('.section-container__card--default');
            if (realCards.length === 0) return;

            // Calculate with a tolerance buffer
            const gridWidth = grid.offsetWidth;
            const cardWidth = realCards[0].offsetWidth;
            const gap = parseInt(getComputedStyle(grid).gap) || 48;

            // Add 20px tolerance to account for rounding/spacing issues
            const tolerance = 20;
            const cardsPerRow = Math.floor((gridWidth + gap + tolerance) / (cardWidth + gap));

            // Don't add placeholders if we are in a single column layout (mobile)
            if (cardsPerRow <= 1) return;

            const remainder = realCards.length % cardsPerRow;
            const placeholdersNeeded = remainder === 0 ? 0 : cardsPerRow - remainder;

            for (let i = 0; i < placeholdersNeeded; i++) {
                const placeholder = document.createElement('div');
                placeholder.className = 'section-container__card section-container__card--placeholder';
                placeholder.setAttribute('aria-hidden', 'true');

            }

        function initFaqAccordion() {

        const questions = document.querySelectorAll('.section-container__faq-wrapper .schema-faq-question');
        
        questions.forEach(question => {
            if (question.dataset.faqInit) return;
            question.dataset.faqInit = 'true';

            const section = question.closest('.schema-faq-section');
            const answer = section ? section.querySelector('.schema-faq-answer') : null;
            const wrapper = question.closest('.section-container__faq-wrapper');

            if (question && answer && wrapper) {
                // Initially hide answers (collapsed state) if not already open
                if (!section.classList.contains('faq-open')) {
                    answer.style.maxHeight = '0';
                    answer.style.overflow = 'hidden';
                }

                // Create click handler
                question._faqClickHandler = function(e) {
                    e.preventDefault();
                    const isOpen = section.classList.contains('faq-open');
                    const allSections = wrapper.querySelectorAll('.schema-faq-section');

                    // Close all other FAQ sections in this wrapper
                    allSections.forEach(s => {
                        if (s !== section) {
                            s.classList.remove('faq-open');
                            const a = s.querySelector('.schema-faq-answer');
                            const q = s.querySelector('.schema-faq-question');
                            if (a) a.style.maxHeight = '0';
                            if (q) q.setAttribute('aria-expanded', 'false');
                        }
                    });

                    // Toggle current section
                    if (!isOpen) {
                        section.classList.add('faq-open');
                        question.setAttribute('aria-expanded', 'true');
                        answer.style.maxHeight = (answer.scrollHeight + 32) + 'px';
                    } else {
                        section.classList.remove('faq-open');
                        question.setAttribute('aria-expanded', 'false');
                        answer.style.maxHeight = '0';
                    }
                };

                question.addEventListener('click', question._faqClickHandler);

                // Make focusable and set ARIA
                question.setAttribute('tabindex', '0');
                question.setAttribute('role', 'button');
                if (!question.hasAttribute('aria-expanded')) {
                    question.setAttribute('aria-expanded', section.classList.contains('faq-open') ? 'true' : 'false');
                }

                question.addEventListener('keydown', function(e) {
                    if (e.target.isContentEditable || e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA') {
                        return;
                    }

                    if (e.key === 'Enter' || e.key === ' ') {
                        e.preventDefault();
                        question.click();
                    } else if (e.key === 'ArrowDown' || e.key === 'ArrowUp') {
                        e.preventDefault();
                        const allQuestions = Array.from(wrapper.querySelectorAll('.schema-faq-question'));
                        const index = allQuestions.indexOf(question);
                        const nextIndex = e.key === 'ArrowDown' ? (index + 1) : (index - 1);
                        if (nextIndex >= 0 && nextIndex < allQuestions.length) {
                            allQuestions[nextIndex].focus();
                        }
                    }
                });
            }

			document.addEventListener('DOMContentLoaded', function () {

				const track = document.querySelector('.mobile-panels-track');
				if (!track) return;

				let currentLevel = 0;

				function slideTo(level) {
					currentLevel = level;
					track.style.transform = 'translateX(-' + (level * 100) + 'vw)';
				}

				function deactivateLevel(level) {
					track.querySelectorAll('.panel[data-level="' + level + '"]').forEach(function (panel) {
						panel.inert = true;
					});
				}

				function activatePanel(panelId) {
					var panel = track.querySelector('.panel[data-panel-id="' + panelId + '"]');
					if (!panel) return null;

					var level = parseInt(panel.dataset.level);

					// Deactivate all panels at this level first
					deactivateLevel(level);

					// Activate the target
					panel.inert = false;

					return panel;
				}

				// Forward navigation
				track.addEventListener('click', function (e) {
					var trigger = e.target.closest('[data-target]');
					if (!trigger) return;

					var targetId = trigger.dataset.target;
					var panel = activatePanel(targetId);
					if (!panel) return;

					var level = parseInt(panel.dataset.level);

					// Update aria-expanded on the trigger
					trigger.setAttribute('aria-expanded', 'true');

					// Slide to that level
					slideTo(level);

					// Move focus after transition
					track.addEventListener('transitionend', function handler() {
						track.removeEventListener('transitionend', handler);
						var focusTarget = panel.querySelector('.panel-back');
						if (focusTarget) focusTarget.focus();
					});
				});

				// Back navigation
				track.addEventListener('click', function (e) {
					var backBtn = e.target.closest('[data-back]');
					if (!backBtn) return;

					var targetLevel = parseInt(backBtn.dataset.back);

					// Inert the panel we're leaving
					var currentPanel = backBtn.closest('.panel');
					if (currentPanel) currentPanel.inert = true;

					// Collapse aria-expanded on the trigger that opened this panel
					var panelId = currentPanel.dataset.panelId;
					var trigger = track.querySelector('[data-target="' + panelId + '"]');
					if (trigger) trigger.setAttribute('aria-expanded', 'false');

					// Slide back
					slideTo(targetLevel);

					// Move focus after transition
					track.addEventListener('transitionend', function handler(evt) {
						if (evt.target !== track) return;
						track.removeEventListener('transitionend', handler);
						var focusTarget = panel.querySelector('.panel-back');
						if (focusTarget) focusTarget.focus();
					});
					
				});

			});
